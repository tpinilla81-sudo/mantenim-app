   100→  marca: string
   101→  modelo: string
   102→  numeroSerie: string
   103→  anioFabricacion: string

   124→  numeroSerie: '',
   125→  anioFabricacion: '',
   126→  ubicacion: '',

   128→  estado: 'OPERATIVO',
   129→  horasUso: '0',
   130→  ultimaRevision: '',
   131→  notas: '',
   132→  clienteId: '',
   133→  revisionesContratadasAnual: '',
   134→  esMaquinaPropia: false,
   135→  horasAnualesEstimadas: '',
   136→  umbralHorasActual: '',
   137→}
   138→
   139→export function EquiposView() {
   140→  // Filtros por columna (tipo Excel autofilter)
   141→  const [colFilters, setColFilters] = useState<Record<string, string>>({})
   142→  const [dialogOpen, setDialogOpen] = useState(false)
   143→  const [editing, setEditing] = useState<Equipo | null>(null)
   144→  const [form, setForm] = useState<EquipoFormData>(EMPTY_FORM)
   145→  const [saving, setSaving] = useState(false)
   146→  const [deleteTarget, setDeleteTarget] = useState<Equipo | null>(null)
   147→  const [uploadingExcel, setUploadingExcel] = useState(false)
   148→  const fileUploadRef = useRef<HTMLInputElement>(null)
   149→  const [toggleTarget, setToggleTarget] = useState<Equipo | null>(null)
   150→  const [deleteAllTarget, setDeleteAllTarget] = useState(false)
   151→
   152→  const { data: clientesResp } = useFetch<{ data: ClienteOption[]; total: number }>(
   153→    '/api/clientes?limit=500'
   154→  )
   155→  const clientesList = clientesResp?.data || []
   156→
   157→  // Cargar TODOS los equipos (sin paginación, con scroll)
   158→  const url = '/api/equipos?limit=2000'
   159→  const { data: resp, loading, error, refetch } = useFetch<{
   160→    data: Equipo[]
   161→    total: number
   162→  }>(url)
   163→
   164→  // Filtrado client-side por columna (como Excel autofilter)
   165→  const allData = resp?.data || []
   166→  const data = useMemo(() => {
   167→    return allData.filter((eq) => {
   168→      for (const [col, val] of Object.entries(colFilters)) {
   169→        if (!val) continue
   170→        const cellVal = String(
   171→          col === 'codigo' ? eq.codigo :
   172→          col === 'modelo' ? eq.modelo :
   173→          col === 'numeroSerie' ? eq.numeroSerie :
   174→          col === 'marcaModelo' ? `${eq.marca} ${eq.modelo}` :
   175→          col === 'esPropio' ? (eq.esMaquinaPropia ? 'Sí' : 'No') :
   176→          col === 'cliente' ? (eq.cliente?.nombre || '') :
   177→          col === 'ubicacion' ? (eq.ubicacion || '') :
   178→          col === 'horasUso' ? String(eq.horasUso) :
   179→          col === 'proximaRevision' ? (eq.proximaRevisionFecha || '') :
   180→          col === 'contrato' ? (eq.contratoActivoId || '') :
   181→          col === 'estado' ? eq.estado :
   182→          ''
   183→        ).toLowerCase()
   184→        if (!cellVal.includes(val.toLowerCase())) return false
   185→      }
   186→      return true
   187→    })
   188→  }, [allData, colFilters])
   189→
   190→  const total = resp?.total || 0
   191→
   192→  const updateFilter = (col: string, val: string) => {
   193→    setColFilters((prev) => {
   194→      const next = { ...prev }
   195→      if (val) next[col] = val
   196→      else delete next[col]
   197→      return next
   198→    })
   199→  }
   200→
   201→  // Siguiente código automático
   202→  useEffect(() => {
   203→    if (dialogOpen && !editing && data && data.length > 0) {
   204→      const maxNum = data.reduce((max, e) => {
   205→        const m = e.codigo.match(/EQ-(\d+)/)
   206→        if (m) return Math.max(max, parseInt(m[1], 10))
   207→        return max
   208→      }, 0)
   209→      const next = `EQ-${String(maxNum + 1).padStart(3, '0')}`
   210→      if (!form.codigo) setForm((f) => ({ ...f, codigo: next }))
   211→    }
   212→  }, [dialogOpen, editing, data])
   213→
   214→  const openCreate = () => {
   215→    setEditing(null)
   216→    setForm({
   217→      ...EMPTY_FORM,
   218→      clienteId: clientesList?.[0]?.id || '',
   219→    })
   220→    setDialogOpen(true)
   221→  }
   222→
   223→  const openEdit = (e: Equipo) => {
   224→    setEditing(e)
   225→    setForm({
   226→      codigo: e.codigo,
   227→      tipo: e.tipo,
   228→      marca: e.marca,
   229→      modelo: e.modelo,
   230→      numeroSerie: e.numeroSerie,
   231→      anioFabricacion: e.anioFabricacion?.toString() || '',
   232→      ubicacion: e.ubicacion || '',
   233→      fechaInstalacion: isoToInputDate(e.fechaInstalacion),
   234→      estado: e.estado,
   235→      horasUso: e.horasUso.toString(),
   236→      ultimaRevision: isoToInputDate(e.ultimaRevision),
   237→      notas: e.notas || '',
   238→      clienteId: e.clienteId,
   239→      revisionesContratadasAnual: e.revisionesContratadasAnual?.toString() || '',
   240→      esMaquinaPropia: e.esMaquinaPropia || false,
   241→      horasAnualesEstimadas: e.horasAnualesEstimadas?.toString() || '',
   242→      umbralHorasActual: e.umbralHorasActual?.toString() || '',
   243→    })
   244→    setDialogOpen(true)
   245→  }
   246→
   247→  const handleSubmit = async (e: React.FormEvent) => {
   248→    e.preventDefault()
   249→    if (!form.codigo.trim() || !form.marca.trim() || !form.modelo.trim() ||
   250→        !form.numeroSerie.trim() || !form.clienteId) {
   251→      toast.error('Completa código, marca, modelo, Nº serie y cliente')
   252→      return
   253→    }
   254→    setSaving(true)
   255→    try {
   256→      // Preparar payload con conversiones de tipos
   257→      const payload = {
   258→        ...form,
   259→        revisionesContratadasAnual: form.revisionesContratadasAnual
   260→          ? Number(form.revisionesContratadasAnual)
   261→          : null,
   262→        horasAnualesEstimadas: form.horasAnualesEstimadas
   263→          ? Number(form.horasAnualesEstimadas)
   264→          : null,
   265→        umbralHorasActual: form.umbralHorasActual
   266→          ? Number(form.umbralHorasActual)
   267→          : null,
   268→      }
   269→      if (editing) {
   270→        await apiFetch(`/api/equipos/${editing.id}`, 'PUT', payload)
   271→        toast.success('Equipo actualizado correctamente')
   272→        // Si cambiaron campos que afectan al cálculo, forzar recalc del equipo
   273→        const cambioRevContratadas = form.revisionesContratadasAnual !== (editing.revisionesContratadasAnual?.toString() || '')
   274→        const cambioUltimaRev = form.ultimaRevision !== (editing.ultimaRevision ? isoToInputDate(editing.ultimaRevision) : '')
   275→        const cambioMaquinaPropia = form.esMaquinaPropia !== (editing.esMaquinaPropia || false)
   276→        const cambioHorasAnuales = form.horasAnualesEstimadas !== (editing.horasAnualesEstimadas?.toString() || '')
   277→        const cambioUmbral = form.umbralHorasActual !== (editing.umbralHorasActual?.toString() || '')
   278→        if (cambioRevContratadas || cambioUltimaRev || cambioMaquinaPropia || cambioHorasAnuales || cambioUmbral) {
   279→          await fetch(`/api/equipos/${editing.id}/recalc-revision`, { method: 'POST' })
   280→            .catch(() => {})
   281→        }
   282→      } else {
   283→        await apiFetch('/api/equipos', 'POST', payload)
   284→        toast.success('Equipo creado correctamente')
   285→      }
   286→      setDialogOpen(false)
   287→      refetch()
   288→    } catch (err) {
   289→      toast.error('Error al guardar', { description: (err as Error).message })
   290→    } finally {
   291→      setSaving(false)
   292→    }
   293→  }
   294→
   295→  const handleDelete = async () => {
   296→    if (!deleteTarget) return
   297→    try {
   298→      await apiFetch(`/api/equipos/${deleteTarget.id}`, 'DELETE')
   299→      toast.success(`Equipo "${deleteTarget.codigo}" eliminado`)
   300→      setDeleteTarget(null)
   301→      refetch()
   302→    } catch (err) {
   303→      toast.error('Error al eliminar', { description: (err as Error).message })
   304→    }
   305→  }
   306→
   307→  const handleDeleteAll = async () => {
   308→    setDeleteAllTarget(false)
   309→    try {
   310→      const res = await fetch('/api/equipos/all', { method: 'DELETE' })
   311→      const json = await res.json()
   312→      if (!res.ok) throw new Error(json.error)
   313→      toast.success(`Listado borrado: ${json.deleted.equipos} equipos eliminados`)
   314→      refetch()
   315→    } catch (err) {
   316→      toast.error('Error al borrar', { description: (err as Error).message })
   317→    }
   318→  }
   319→
   320→  const handleToggleRevision = async (eq: Equipo) => {
   321→    try {
   322→      const res = await fetch(`/api/equipos/${eq.id}/toggle-revision`, { method: 'POST' })
   323→      const json = await res.json()
   324→      if (!res.ok) throw new Error(json.error)
   325→      toast.success(
   326→        `${eq.codigo} → ${json.nuevoTipo === 'POR_CONTRATO' ? 'Por contrato (M1/M2)' : 'Por horas'}`
   327→      )
   328→      refetch()
   329→    } catch (err) {
   330→      toast.error('Error al alternar tipo de revisión', { description: (err as Error).message })
   331→    }
   332→  }
   333→
   334→  return (
   335→    <div className="p-4 lg:p-8 space-y-5 max-w-[1400px]">
   336→      {/* Toolbar */}
   337→      <div className="flex flex-col lg:flex-row lg:items-center gap-3">
   338→        <p className="text-[13px] text-muted-foreground">
   339→          <strong className="text-foreground tabular-nums">{data.length}</strong>
   340→          {data.length !== total ? ` de ${total}` : ''} equipos
   341→          {Object.keys(colFilters).length > 0 && ' (filtrados)'}
   342→        </p>
   343→        <div className="flex-1" />
   344→        <a href="/api/equipos/template" download>
   345→          <Button variant="outline" size="sm">
   346→            <FileSpreadsheet className="w-4 h-4 mr-1.5" />
   347→            Descargar plantilla
   348→          </Button>
   349→        </a>
   350→        <Button
   351→          variant="outline"
   352→          size="sm"
   353→          onClick={() => fileUploadRef.current?.click()}
   354→          disabled={uploadingExcel}
   355→        >
   356→          {uploadingExcel ? (
   357→            <>
   358→              <Loader2 className="w-4 h-4 mr-1.5 animate-spin" />
   359→              Subiendo...
   360→            </>
   361→          ) : (
   362→            <>
   363→              <Upload className="w-4 h-4 mr-1.5" />
   364→              Subir Excel
   365→            </>
   366→          )}
   367→        </Button>
   368→        <input
   369→          ref={fileUploadRef}
   370→          type="file"
   371→          accept=".xlsx,.xls"
   372→          className="hidden"
   373→          onChange={async (e) => {
   374→            const file = e.target.files?.[0]
   375→            if (!file) return
   376→            setUploadingExcel(true)
   377→            try {
   378→              const formData = new FormData()
   379→              formData.append('file', file)
   380→              formData.append('dryRun', 'false')
   381→              const res = await fetch('/api/sync/excel-flexible', {
   382→                method: 'POST',
   383→                body: formData,
   384→              })
   385→              const json = await res.json()
   386→              if (!res.ok) throw new Error(json.error || json.detail || `Error ${res.status}`)
   387→              toast.success(
   388→                `Excel procesado: ${json.rowsUpdated || 0} actualizados, ${json.rowsTotal || 0} filas total`,
   389→                { description: `${json.columnsDetected?.length || 0} columnas reconocidas` }
   390→              )
   391→              refetch()
   392→            } catch (err) {
   393→              toast.error('Error al subir Excel', { description: (err as Error).message })
   394→            } finally {
   395→              setUploadingExcel(false)
   396→              if (fileUploadRef.current) fileUploadRef.current.value = ''
   397→            }
   398→          }}
   399→        />
   400→        <Button onClick={openCreate} disabled={!clientesList?.length}>
   401→          <Plus className="w-4 h-4 mr-1.5" />
   402→          Nuevo equipo
   403→        </Button>
   404→        {data.length > 0 && (
   405→          <Button
   406→            variant="outline"
   407→            size="sm"
   408→            onClick={() => setDeleteAllTarget(true)}
   409→            className="text-rose-600 hover:text-rose-700 hover:bg-rose-50 border-rose-200"
   410→          >
   411→            <Trash2 className="w-3.5 h-3.5 mr-1.5" />
   412→            Borrar listado
   413→          </Button>
   414→        )}
   415→      </div>
   416→
   417→      {/* Tabla */}
   418→      <Card className="border-border/80">
   419→        <CardContent className="p-0 overflow-hidden">
   420→          {loading ? (
   421→            <div className="p-8 space-y-3">
   422→              {Array.from({ length: 5 }).map((_, i) => (
   423→                <div key={i} className="h-14 bg-muted/50 rounded animate-pulse" />
   424→              ))}
   425→            </div>
   426→          ) : error ? (
   427→            <div className="p-8 text-rose-700 text-sm">Error: {error}</div>
   428→          ) : data.length === 0 ? (
   429→            <div className="p-12 text-center">
   430→              <Forklift className="w-10 h-10 text-muted-foreground mx-auto mb-3" />
   431→              <p className="text-sm font-medium">No hay equipos que mostrar</p>
   432→              <p className="text-[12px] text-muted-foreground mt-1">
   433→                Registra una grúa o montacargas para empezar.
   434→              </p>
   435→            </div>
   436→          ) : (
   437→            <div className="overflow-auto scrollbar-thin relative" style={{ maxHeight: 'calc(100vh - 200px)' }}>
   438→              <Table>
   439→                <TableHeader>
   440→                  <TableRow className="bg-muted/60 hover:bg-muted/60 sticky top-0 z-10">
   441→                    <TableHead className="min-w-[90px]">
   442→                      Nº
   443→                      <Input
   444→                        value={colFilters.codigo || ''}
   445→                        onChange={(e) => updateFilter('codigo', e.target.value)}
   446→                        placeholder="Filtrar..."
   447→                        className="h-6 mt-1 text-[11px]"
   448→                      />
   449→                    </TableHead>
   450→                    <TableHead className="min-w-[110px]">
   451→                      Descripción
   452→                      <Input
   453→                        value={colFilters.modelo || ''}
   454→                        onChange={(e) => updateFilter('modelo', e.target.value)}
   455→                        placeholder="Filtrar..."
   456→                        className="h-6 mt-1 text-[11px]"
   457→                      />
   458→                    </TableHead>
   459→                    <TableHead className="min-w-[110px]">
   460→                      Nº Serie
   461→                      <Input
   462→                        value={colFilters.numeroSerie || ''}
   463→                        onChange={(e) => updateFilter('numeroSerie', e.target.value)}
   464→                        placeholder="Filtrar..."
   465→                        className="h-6 mt-1 text-[11px]"
   466→                      />
   467→                    </TableHead>
   468→                    <TableHead className="min-w-[110px]">
   469→                      Modelo
   470→                      <Input
   471→                        value={colFilters.marcaModelo || ''}
   472→                        onChange={(e) => updateFilter('marcaModelo', e.target.value)}
   473→                        placeholder="Filtrar..."
   474→                        className="h-6 mt-1 text-[11px]"
   475→                      />
   476→                    </TableHead>
   477→                    <TableHead className="min-w-[80px] text-center">
   478→                      Es propio
   479→                      <select
   480→                        value={colFilters.esPropio || ''}
   481→                        onChange={(e) => updateFilter('esPropio', e.target.value)}
   482→                        className="h-6 mt-1 text-[11px] w-full rounded border border-input bg-background px-1"
   483→                      >
   484→                        <option value="">Todos</option>
   485→                        <option value="Sí">Sí</option>
   486→                        <option value="No">No</option>
   487→                      </select>
   488→                    </TableHead>
   489→                    <TableHead className="min-w-[140px]">
   490→                      Cliente
   491→                      <Input
   492→                        value={colFilters.cliente || ''}
   493→                        onChange={(e) => updateFilter('cliente', e.target.value)}
   494→                        placeholder="Filtrar..."
   495→                        className="h-6 mt-1 text-[11px]"
   496→                      />
   497→                    </TableHead>
   498→                    <TableHead className="min-w-[100px]">
   499→                      Ubicación
   500→                      <Input
   501→                        value={colFilters.ubicacion || ''}
   502→                        onChange={(e) => updateFilter('ubicacion', e.target.value)}
   503→                        placeholder="Filtrar..."
   504→                        className="h-6 mt-1 text-[11px]"
   505→                      />
   506→                    </TableHead>
   507→                    <TableHead className="text-center min-w-[70px]">
   508→                      Horas
   509→                      <Input
   510→                        value={colFilters.horasUso || ''}
   511→                        onChange={(e) => updateFilter('horasUso', e.target.value)}
   512→                        placeholder="Filtrar..."
   513→                        className="h-6 mt-1 text-[11px]"
   514→                      />
   515→                    </TableHead>
   516→                    <TableHead className="min-w-[100px]">
   517→                      Próx. rev.
   518→                      <Input
   519→                        value={colFilters.proximaRevision || ''}
   520→                        onChange={(e) => updateFilter('proximaRevision', e.target.value)}
   521→                        placeholder="Filtrar..."
   522→                        className="h-6 mt-1 text-[11px]"
   523→                      />
   524→                    </TableHead>
   525→                    <TableHead className="min-w-[100px]">
   526→                      Contrato
   527→                      <Input
   528→                        value={colFilters.contrato || ''}
   529→                        onChange={(e) => updateFilter('contrato', e.target.value)}
   530→                        placeholder="Filtrar..."
   531→                        className="h-6 mt-1 text-[11px]"
   532→                      />
   533→                    </TableHead>
   534→                    <TableHead className="text-center min-w-[90px]">
   535→                      Estado
   536→                      <select
   537→                        value={colFilters.estado || ''}
   538→                        onChange={(e) => updateFilter('estado', e.target.value)}
   539→                        className="h-6 mt-1 text-[11px] w-full rounded border border-input bg-background px-1"
   540→                      >
   541→                        <option value="">Todos</option>
   542→                        <option value="OPERATIVO">Operativo</option>
   543→                        <option value="EN_MANTENIMIENTO">En mant.</option>
