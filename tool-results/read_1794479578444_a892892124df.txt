   139→   139→export function EquiposView() {
   140→   140→  // Filtros por columna (tipo Excel autofilter)
   141→   141→  const [colFilters, setColFilters] = useState<Record<string, string>>({})
   142→   142→  const [dialogOpen, setDialogOpen] = useState(false)
   143→   143→  const [editing, setEditing] = useState<Equipo | null>(null)
   144→   144→  const [form, setForm] = useState<EquipoFormData>(EMPTY_FORM)
   145→   145→  const [saving, setSaving] = useState(false)
   146→   146→  const [deleteTarget, setDeleteTarget] = useState<Equipo | null>(null)
   147→   147→  const [uploadingExcel, setUploadingExcel] = useState(false)
   148→   148→  const fileUploadRef = useRef<HTMLInputElement>(null)
   149→   149→  const [toggleTarget, setToggleTarget] = useState<Equipo | null>(null)
   150→   150→  const [deleteAllTarget, setDeleteAllTarget] = useState(false)
   151→   151→
   152→   152→  const { data: clientesResp } = useFetch<{ data: ClienteOption[]; total: number }>(
   153→   153→    '/api/clientes?limit=500'
   154→   154→  )
   155→   155→  const clientesList = clientesResp?.data || []
   156→   156→
   157→   157→  // Cargar TODOS los equipos (sin paginación, con scroll)
   158→   158→  const url = '/api/equipos?limit=2000'
   159→   159→  const { data: resp, loading, error, refetch } = useFetch<{
   160→   160→    data: Equipo[]
   161→   161→    total: number
   162→   162→  }>(url)
   163→   163→
   164→   164→  // Filtrado client-side por columna (como Excel autofilter)
   165→   165→  const allData = resp?.data || []
   166→   166→  const data = useMemo(() => {
   167→   167→    return allData.filter((eq) => {
   168→   168→      for (const [col, val] of Object.entries(colFilters)) {
   169→   169→        if (!val) continue
   170→   170→        const cellVal = String(
   171→   171→          col === 'codigo' ? eq.codigo :
   172→   172→          col === 'modelo' ? eq.modelo :
   173→   173→          col === 'numeroSerie' ? eq.numeroSerie :
   174→   174→          col === 'marcaModelo' ? `${eq.marca} ${eq.modelo}` :
   175→   175→          col === 'esPropio' ? (eq.esMaquinaPropia ? 'Sí' : 'No') :
   176→   176→          col === 'cliente' ? (eq.cliente?.nombre || '') :
   177→   177→          col === 'ubicacion' ? (eq.ubicacion || '') :
   178→   178→          col === 'horasUso' ? String(eq.horasUso) :
   179→   179→          col === 'proximaRevision' ? (eq.proximaRevisionFecha || '') :
   180→   180→          col === 'contrato' ? (eq.contratoActivoId || '') :
   181→   181→          col === 'estado' ? eq.estado :
   182→   182→          ''
   183→   183→        ).toLowerCase()
   184→   184→        if (!cellVal.includes(val.toLowerCase())) return false
   185→   185→      }
   186→   186→      return true
   187→   187→    })
   188→   188→  }, [allData, colFilters])
   189→   189→
   190→   190→  const total = resp?.total || 0
   191→   191→
   192→   192→  const updateFilter = (col: string, val: string) => {
   193→   193→    setColFilters((prev) => {
   194→   194→      const next = { ...prev }
   195→   195→      if (val) next[col] = val
   196→   196→      else delete next[col]
   197→   197→      return next
   198→   198→    })
   199→   199→  }
   200→   200→
   201→   201→  // Siguiente código automático
   202→   202→  useEffect(() => {
   203→   203→    if (dialogOpen && !editing && data && data.length > 0) {
   204→   204→      const maxNum = data.reduce((max, e) => {
   205→   205→        const m = e.codigo.match(/EQ-(\d+)/)
   206→   206→        if (m) return Math.max(max, parseInt(m[1], 10))
   207→   207→        return max
   208→   208→      }, 0)
   209→   209→      const next = `EQ-${String(maxNum + 1).padStart(3, '0')}`
   210→   210→      if (!form.codigo) setForm((f) => ({ ...f, codigo: next }))
   211→   211→    }
   212→   212→  }, [dialogOpen, editing, data])
   213→   213→
   214→   214→  const openCreate = () => {
   215→   215→    setEditing(null)
   216→   216→    setForm({
   217→   217→      ...EMPTY_FORM,
   218→   218→      clienteId: clientesList?.[0]?.id || '',
   219→   219→    })
   220→   220→    setDialogOpen(true)
   221→   221→  }
   222→   222→
   223→   223→  const openEdit = (e: Equipo) => {
   224→   224→    setEditing(e)
   225→   225→    setForm({
   226→   226→      codigo: e.codigo,
   227→   227→      tipo: e.tipo,
   228→   228→      marca: e.marca,
   229→   229→      modelo: e.modelo,
   230→   230→      numeroSerie: e.numeroSerie,
   231→   231→      anioFabricacion: e.anioFabricacion?.toString() || '',
   232→   232→      ubicacion: e.ubicacion || '',
   233→   233→      fechaInstalacion: isoToInputDate(e.fechaInstalacion),
   234→   234→      estado: e.estado,
   235→   235→      horasUso: e.horasUso.toString(),
   236→   236→      ultimaRevision: isoToInputDate(e.ultimaRevision),
   237→   237→      notas: e.notas || '',
   238→   238→      clienteId: e.clienteId,
   239→   239→      revisionesContratadasAnual: e.revisionesContratadasAnual?.toString() || '',
   240→   240→      esMaquinaPropia: e.esMaquinaPropia || false,
   241→   241→      horasAnualesEstimadas: e.horasAnualesEstimadas?.toString() || '',
   242→   242→      umbralHorasActual: e.umbralHorasActual?.toString() || '',
   243→   243→    })
   244→   244→    setDialogOpen(true)
   245→   245→  }
   246→   246→
   247→   247→  const handleSubmit = async (e: React.FormEvent) => {
   248→   248→    e.preventDefault()
   249→   249→    if (!form.codigo.trim() || !form.marca.trim() || !form.modelo.trim() ||
   250→   250→        !form.numeroSerie.trim() || !form.clienteId) {
   251→   251→      toast.error('Completa código, marca, modelo, Nº serie y cliente')
   252→   252→      return
   253→   253→    }
   254→   254→    setSaving(true)
   255→   255→    try {
   256→   256→      // Preparar payload con conversiones de tipos
   257→   257→      const payload = {
   258→   258→        ...form,
   259→   259→        revisionesContratadasAnual: form.revisionesContratadasAnual
   260→   260→          ? Number(form.revisionesContratadasAnual)
   261→   261→          : null,
   262→   262→        horasAnualesEstimadas: form.horasAnualesEstimadas
   263→   263→          ? Number(form.horasAnualesEstimadas)
   264→   264→          : null,
   265→   265→        umbralHorasActual: form.umbralHorasActual
   266→   266→          ? Number(form.umbralHorasActual)
   267→   267→          : null,
   268→   268→      }
   269→   269→      if (editing) {
   270→   270→        await apiFetch(`/api/equipos/${editing.id}`, 'PUT', payload)
   271→   271→        toast.success('Equipo actualizado correctamente')
   272→   272→        // Si cambiaron campos que afectan al cálculo, forzar recalc del equipo
   273→   273→        const cambioRevContratadas = form.revisionesContratadasAnual !== (editing.revisionesContratadasAnual?.toString() || '')
   274→   274→        const cambioUltimaRev = form.ultimaRevision !== (editing.ultimaRevision ? isoToInputDate(editing.ultimaRevision) : '')
   275→   275→        const cambioMaquinaPropia = form.esMaquinaPropia !== (editing.esMaquinaPropia || false)
   276→   276→        const cambioHorasAnuales = form.horasAnualesEstimadas !== (editing.horasAnualesEstimadas?.toString() || '')
   277→   277→        const cambioUmbral = form.umbralHorasActual !== (editing.umbralHorasActual?.toString() || '')
   278→   278→        if (cambioRevContratadas || cambioUltimaRev || cambioMaquinaPropia || cambioHorasAnuales || cambioUmbral) {
   279→   279→          await fetch(`/api/equipos/${editing.id}/recalc-revision`, { method: 'POST' })
   280→   280→            .catch(() => {})
   281→   281→        }
   282→   282→      } else {
   283→   283→        await apiFetch('/api/equipos', 'POST', payload)
   284→   284→        toast.success('Equipo creado correctamente')
   285→   285→      }
   286→   286→      setDialogOpen(false)
   287→   287→      refetch()
   288→   288→    } catch (err) {
   289→   289→      toast.error('Error al guardar', { description: (err as Error).message })
   290→   290→    } finally {
   291→   291→      setSaving(false)
   292→   292→    }
   293→   293→  }
   294→   294→
   295→   295→  const handleDelete = async () => {
   296→   296→    if (!deleteTarget) return
   297→   297→    try {
   298→   298→      await apiFetch(`/api/equipos/${deleteTarget.id}`, 'DELETE')
   299→   299→      toast.success(`Equipo "${deleteTarget.codigo}" eliminado`)
   300→   300→      setDeleteTarget(null)
   301→   301→      refetch()
   302→   302→    } catch (err) {
   303→   303→      toast.error('Error al eliminar', { description: (err as Error).message })
   304→   304→    }
   305→   305→  }
   306→   306→
   307→   307→  const handleDeleteAll = async () => {
   308→   308→    setDeleteAllTarget(false)
   309→   309→    try {
   310→   310→      const res = await fetch('/api/equipos/all', { method: 'DELETE' })
   311→   311→      const json = await res.json()
   312→   312→      if (!res.ok) throw new Error(json.error)
   313→   313→      toast.success(`Listado borrado: ${json.deleted.equipos} equipos eliminados`)
   314→   314→      refetch()
   315→   315→    } catch (err) {
   316→   316→      toast.error('Error al borrar', { description: (err as Error).message })
   317→   317→    }
   318→   318→  }
   319→   319→
   320→   320→  const handleToggleRevision = async (eq: Equipo) => {
   321→   321→    try {
   322→   322→      const res = await fetch(`/api/equipos/${eq.id}/toggle-revision`, { method: 'POST' })
   323→   323→      const json = await res.json()
   324→   324→      if (!res.ok) throw new Error(json.error)
   325→   325→      toast.success(
   326→   326→        `${eq.codigo} → ${json.nuevoTipo === 'POR_CONTRATO' ? 'Por contrato (M1/M2)' : 'Por horas'}`
   327→   327→      )
   328→   328→      refetch()
   329→   329→    } catch (err) {
   330→   330→      toast.error('Error al alternar tipo de revisión', { description: (err as Error).message })
   331→   331→    }
   332→   332→  }
   333→   333→
   334→   334→  return (
   335→   335→    <div className="p-4 lg:p-8 space-y-5 max-w-[1400px]">
   336→   336→      {/* Toolbar */}
   337→   337→      <div className="flex flex-col lg:flex-row lg:items-center gap-3">
   338→   338→        <p className="text-[13px] text-muted-foreground">
   339→   339→          <strong className="text-foreground tabular-nums">{data.length}</strong>
   340→   340→          {data.length !== total ? ` de ${total}` : ''} equipos
   341→   341→          {Object.keys(colFilters).length > 0 && ' (filtrados)'}
   342→   342→        </p>
   343→   343→        <div className="flex-1" />
   344→   344→        <a href="/api/equipos/template" download>
   345→   345→          <Button variant="outline" size="sm">
   346→   346→            <FileSpreadsheet className="w-4 h-4 mr-1.5" />
   347→   347→            Descargar plantilla
   348→   348→          </Button>
   349→   349→        </a>
   350→   350→        <Button
   351→   351→          variant="outline"
   352→   352→          size="sm"
   353→   353→          onClick={() => fileUploadRef.current?.click()}
   354→   354→          disabled={uploadingExcel}
   355→   355→        >
   356→   356→          {uploadingExcel ? (
   357→   357→            <>
   358→   358→              <Loader2 className="w-4 h-4 mr-1.5 animate-spin" />
   359→   359→              Subiendo...
   360→   360→            </>
   361→   361→          ) : (
   362→   362→            <>
   363→   363→              <Upload className="w-4 h-4 mr-1.5" />
   364→   364→              Subir Excel
   365→   365→            </>
   366→   366→          )}
   367→   367→        </Button>
   368→   368→        <input
   369→   369→          ref={fileUploadRef}
   370→   370→          type="file"
   371→   371→          accept=".xlsx,.xls"
   372→   372→          className="hidden"
   373→   373→          onChange={async (e) => {
   374→   374→            const file = e.target.files?.[0]
   375→   375→            if (!file) return
   376→   376→            setUploadingExcel(true)
   377→   377→            try {
   378→   378→              const formData = new FormData()
   379→   379→              formData.append('file', file)
   380→   380→              formData.append('dryRun', 'false')
   381→   381→              const res = await fetch('/api/sync/excel-flexible', {
   382→   382→                method: 'POST',
   383→   383→                body: formData,
   384→   384→              })
   385→   385→              const json = await res.json()
   386→   386→              if (!res.ok) throw new Error(json.error || json.detail || `Error ${res.status}`)
   387→   387→              toast.success(
   388→   388→                `Excel procesado: ${json.rowsUpdated || 0} actualizados, ${json.rowsTotal || 0} filas total`,
   389→   389→                { description: `${json.columnsDetected?.length || 0} columnas reconocidas` }
   390→   390→              )
   391→   391→              refetch()
   392→   392→            } catch (err) {
   393→   393→              toast.error('Error al subir Excel', { description: (err as Error).message })
   394→   394→            } finally {
   395→   395→              setUploadingExcel(false)
   396→   396→              if (fileUploadRef.current) fileUploadRef.current.value = ''
   397→   397→            }
   398→   398→          }}
   399→   399→        />
   400→   400→        <Button onClick={openCreate} disabled={!clientesList?.length}>
   401→   401→          <Plus className="w-4 h-4 mr-1.5" />
   402→   402→          Nuevo equipo
   403→   403→        </Button>
   404→   404→        {data.length > 0 && (
   405→   405→          <Button
   406→   406→            variant="outline"
   407→   407→            size="sm"
   408→   408→            onClick={() => setDeleteAllTarget(true)}
   409→   409→            className="text-rose-600 hover:text-rose-700 hover:bg-rose-50 border-rose-200"
   410→   410→          >
   411→   411→            <Trash2 className="w-3.5 h-3.5 mr-1.5" />
   412→   412→            Borrar listado
   413→   413→          </Button>
   414→   414→        )}
   415→   415→      </div>
   416→   416→
   417→   417→      {/* Tabla */}
   418→   418→      <Card className="border-border/80">
   419→   419→        <CardContent className="p-0 overflow-hidden">
   420→   420→          {loading ? (
   421→   421→            <div className="p-8 space-y-3">
   422→   422→              {Array.from({ length: 5 }).map((_, i) => (
   423→   423→                <div key={i} className="h-14 bg-muted/50 rounded animate-pulse" />
   424→   424→              ))}
   425→   425→            </div>
   426→   426→          ) : error ? (
   427→   427→            <div className="p-8 text-rose-700 text-sm">Error: {error}</div>
   428→   428→          ) : data.length === 0 ? (
   429→   429→            <div className="p-12 text-center">
   430→   430→              <Forklift className="w-10 h-10 text-muted-foreground mx-auto mb-3" />
   431→   431→              <p className="text-sm font-medium">No hay equipos que mostrar</p>
   432→   432→              <p className="text-[12px] text-muted-foreground mt-1">
   433→   433→                Registra una grúa o montacargas para empezar.
   434→   434→              </p>
   435→   435→            </div>
   436→   436→          ) : (
   437→   437→            <div className="overflow-auto scrollbar-thin relative" style={{ maxHeight: 'calc(100vh - 200px)' }}>
   438→   438→              <Table>
   439→   439→                <TableHeader>
   440→   440→                  <TableRow className="bg-muted/60 hover:bg-muted/60 sticky top-0 z-10">
   441→   441→                    <TableHead className="min-w-[90px]">
   442→   442→                      Nº
   443→   443→                      <Input
   444→   444→                        value={colFilters.codigo || ''}
   445→   445→                        onChange={(e) => updateFilter('codigo', e.target.value)}
   446→   446→                        placeholder="Filtrar..."
   447→   447→                        className="h-6 mt-1 text-[11px]"
   448→   448→                      />
   449→   449→                    </TableHead>
   450→   450→                    <TableHead className="min-w-[110px]">
   451→   451→                      Descripción
   452→   452→                      <Input
