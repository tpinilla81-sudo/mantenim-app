    95→function formatFileSize(bytes: number): string {
    96→  if (bytes < 1024) return `${bytes} B`
    97→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
    98→  return `${(bytes / (1024 * 1024)).toFixed(2)} MB`
    99→}
   100→
   101→function formatDuration(ms: number): string {
   102→  if (ms < 1000) return `${ms} ms`
   103→  if (ms < 60000) return `${(ms / 1000).toFixed(1)} s`
   104→  const m = Math.floor(ms / 60000)
   105→  const s = Math.round((ms % 60000) / 1000)
   106→  return `${m}m ${s}s`
   107→}

   606→                variant="ghost"

   608→              >

   612→            <div className="flex-1" />
   613→            <Button
   614→              variant="outline"
   615→              onClick={handleRecalc}

   638→              <Trash2 className="w-3.5 h-3.5 mr-1.5" />

   640→            </Button>
   641→          </div>
   642→
   643→          {/* Barra de progreso */}
   644→          {progress > 0 && (

   653→                {progress < 40 ? 'Subiendo archivo...' :

   657→              </p>

   659→          )}
   660→

   667→                  Sincronización completada

   670→              <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   671→                <ResultMetric label="Clientes nuevos" value={lastResult.clientesNuevos} color="emerald" />

   674→                <ResultMetric label="Filas omitidas" value={lastResult.rowsSkipped} color="gray" />

   676→                  <ResultMetric label="Filas con error" value={lastResult.rowsFailed} color="rose" />
   677→                )}
   678→              </div>
   679→              <div className="text-[12px] text-muted-foreground">
   680→                Tiempo: {formatDuration(lastResult.duration)} · Archivo: {lastResult.fileName}
   681→              </div>

   694→            </div>
   695→          )}
   696→        </CardContent>

   698→
   699→      {/* Historial de sincronizaciones */}
   700→      <Card>

   705→          </CardTitle>

   711→            </p>

   746→                        {log.contratosNuevos}

   750→                      </td>

   755→                          <span className="text-muted-foreground">0</span>

   759→                        <EstadoBadge estado={log.status} label={

   763→                        } />
   764→                      </td>
   765→                      <td className="px-2 py-2 text-right text-[12px] text-muted-foreground tabular-nums">
   766→                        {formatDuration(log.duracionMs)}
   767→                      </td>
   768→                    </tr>

   770→                </tbody>
   771→              </table>
   772→            </div>
   773→          )}
   774→        </CardContent>
   775→      </Card>
   776→
   777→      <ConfirmDialog

   783→        onConfirm={handleDeleteAll}
   784→        onCancel={() => setDeleteTarget(false)}
   785→      />
   786→    </div>

   790→function Metric({ label, value, icon: Icon, color }: {
   791→  label: string
   792→  value: number
   793→  icon: React.ComponentType<{ className?: string }>
   794→  color?: 'emerald' | 'amber' | 'sky' | 'gray'
   795→}) {
   796→  const colorClass = {
   797→    emerald: 'bg-emerald-50 text-emerald-700',
   798→    amber: 'bg-amber-50 text-amber-700',
   799→    sky: 'bg-sky-50 text-sky-700',
   800→    gray: 'bg-muted text-muted-foreground',
   801→  }[color || 'gray']
   802→  return (
   803→    <div className="flex items-center gap-2">
   804→      <div className={cn('w-8 h-8 rounded-md flex items-center justify-center', colorClass)}>
   805→        <Icon className="w-4 h-4" />
   806→      </div>
   807→      <div>
   808→        <p className="text-[11px] text-muted-foreground">{label}</p>
   809→        <p className="text-sm font-semibold tabular-nums">{value.toLocaleString('es-ES')}</p>
   810→      </div>
   811→    </div>
   812→  )
   813→}

   815→function ResultMetric({ label, value, color }: { label: string; value: number; color: 'emerald' | 'sky' | 'gray' | 'rose' }) {

   826→      </p>
